// ── Import the host-side bridge library ──────────────────────────────────────
// Served from the same local static server as the plugin (port 8080).
// Adjust path if you move codap-quarto-bridge.js elsewhere.
CodapBridgeModule = import("http://localhost:8080/codap-quarto-bridge.js")
mutable bridgeStatusText = "⏳ Waiting for bridge module…"
mutable componentListText = "(not yet fetched)"
mutable eventLogText = "(no events yet)"
mutable eventCount = 0
mutable calcResultText = "(not yet fetched)"
// ── Connect and run the round-trip test ──────────────────────────────────────
{
  const { createBridge } = await CodapBridgeModule;

  mutable bridgeStatusText = "⏳ Connecting to CODAP bridge plugin…";

  let bridge;
  try {
    bridge = await createBridge({
      timeout: 8000,
      handshakeTimeout: 12000,
    });
  } catch (err) {
    mutable bridgeStatusText = `❌ Connection failed:\n${err.message}\n\nCheck that:\n  • CODAP is running on localhost:8081\n  • Plugin is served on localhost:8080\n  • The plugin has finished initializing in CODAP`;
    return;
  }

  mutable bridgeStatusText = "✅ Connected to CODAP bridge plugin";

  // ── Round-trip: get componentList ─────────────────────────────────────────
  // This is the safest first resource to read — always exists in any CODAP doc.
  // Once you know your calculator component name, swap in:
  //   bridge.get("component[YOUR_CALC_NAME]")
// ── Round-trip: get componentList ────────────────────────────────────────
  try {
    mutable bridgeStatusText = "✅ Connected — fetching componentList…";
    const result = await bridge.get("componentList");
    console.log("[diag] componentList raw:", JSON.stringify(result, null, 2));
    mutable componentListText = JSON.stringify(result, null, 2);
    mutable bridgeStatusText = `✅ Round-trip complete — ${Array.isArray(result) ? result.length : "?"} component(s) found`;

    // ── Targeted Calculator read (uses same result, no extra request) ───────
    const ids = Array.isArray(result) ? result : (result?.values ?? []);
    console.log("[diag] component ids/types:", ids);
    // Once you see the id in the console, swap in: bridge.get(`component[${id}]`)

  } catch (err) {
    mutable componentListText = `Error: ${err.message}`;
    mutable bridgeStatusText = `⚠️ Connected but get() failed:\n${err.message}`;
  }
  
  // ── Subscribe to live CODAP events ────────────────────────────────────────
  bridge.subscribe((payload) => {
    mutable eventCount = eventCount + 1;
    const line = `[${new Date().toLocaleTimeString()}] #${eventCount} ${payload?.eventType ?? "unknown"}: ${JSON.stringify(payload).slice(0, 120)}`;
    console.log("[quarto-bridge] event:", payload);
    mutable eventLogText = line + "\n" + eventLogText;
  });
}

Next Steps

Once the round-trip above works:

  1. Find your calculator component name — look at the componentList output above. Find the entry with "type": "Calculator" and note its name or id.

  2. Swap in a targeted get:

    bridge.get("component[YOUR_CALC_NAME]")

  3. Wire up a reactive OJS cell — use bridge.observable("component[NAME]", "poll") to get a value that updates automatically.

  4. Inline display — once you have a reactive value, you can reference it in Quarto markdown with an OJS inline expression: ${calcValue}.
